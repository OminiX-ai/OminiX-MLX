```rust
//! Gemma 4 text config parsing.
use serde::Deserialize;
use crate::error::{Error, Result};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LayerKind { Sliding, Global }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum RopeType { Default, Proportional }

#[derive(Debug, Clone, Copy)]
pub struct RopeSpec {
    pub rope_type: RopeType,
    pub theta: f32,
    /// 仅 proportional 有意义;default 取 1.0(全维旋转)。
    pub partial_rotary_factor: f32,
}

#[derive(Debug, Clone)]
pub struct ModelArgs {
    pub num_hidden_layers: i32,
    pub hidden_size: i32,
    pub num_attention_heads: i32,
    pub num_key_value_heads: i32,
    pub head_dim: i32,
    pub global_head_dim: i32,
    pub num_global_key_value_heads: i32,
    pub attention_k_eq_v: bool,
    pub intermediate_size: i32,
    pub vocab_size: i32,
    pub tie_word_embeddings: bool,
    pub rms_norm_eps: f32,
    pub final_logit_softcapping: f32,
    pub sliding_window: i32,
    pub max_position_embeddings: i32,
    pub layer_types: Vec<LayerKind>,
    pub rope_sliding: RopeSpec,
    pub rope_global: RopeSpec,
}

// ---- serde 中间结构 ----
#[derive(Deserialize)]
struct Root { text_config: TextConfig }

#[derive(Deserialize)]
struct TextConfig {
    num_hidden_layers: i32,
    hidden_size: i32,
    num_attention_heads: i32,
    num_key_value_heads: i32,
    head_dim: i32,
    global_head_dim: i32,
    num_global_key_value_heads: i32,
    #[serde(default)] attention_k_eq_v: bool,
    intermediate_size: i32,
    vocab_size: i32,
    tie_word_embeddings: bool,
    rms_norm_eps: f32,
    #[serde(default)] final_logit_softcapping: f32,
    sliding_window: i32,
    max_position_embeddings: i32,
    layer_types: Vec<String>,
    rope_parameters: RopeParameters,
}

#[derive(Deserialize)]
struct RopeParameters { full_attention: RopeRaw, sliding_attention: RopeRaw }

#[derive(Deserialize)]
struct RopeRaw {
    rope_type: String,
    rope_theta: f32,
    #[serde(default = "default_partial")] partial_rotary_factor: f32,
}
fn default_partial() -> f32 { 1.0 }

impl ModelArgs {
    pub fn from_config_str(s: &str) -> Result<Self> {
        let root: Root = serde_json::from_str(s)?;
        let t = root.text_config;

        let layer_types = t.layer_types.iter().map(|s| match s.as_str() {
            "sliding_attention" => Ok(LayerKind::Sliding),
            "full_attention" => Ok(LayerKind::Global),
            other => Err(Error::Config(format!("unknown layer_type {other}"))),
        }).collect::<Result<Vec<_>>>()?;

        let parse_rope = |r: &RopeRaw| -> Result<RopeSpec> {
            let rope_type = match r.rope_type.as_str() {
                "default" | "linear" => RopeType::Default,
                "proportional" => RopeType::Proportional,
                other => return Err(Error::Config(format!("unknown rope_type {other}"))),
            };
            Ok(RopeSpec { rope_type, theta: r.rope_theta, partial_rotary_factor: r.partial_rotary_factor })
        };

        Ok(ModelArgs {
            num_hidden_layers: t.num_hidden_layers,
            hidden_size: t.hidden_size,
            num_attention_heads: t.num_attention_heads,
            num_key_value_heads: t.num_key_value_heads,
            head_dim: t.head_dim,
            global_head_dim: t.global_head_dim,
            num_global_key_value_heads: t.num_global_key_value_heads,
            attention_k_eq_v: t.attention_k_eq_v,
            intermediate_size: t.intermediate_size,
            vocab_size: t.vocab_size,
            tie_word_embeddings: t.tie_word_embeddings,
            rms_norm_eps: t.rms_norm_eps,
            final_logit_softcapping: t.final_logit_softcapping,
            sliding_window: t.sliding_window,
            max_position_embeddings: t.max_position_embeddings,
            layer_types,
            rope_sliding: parse_rope(&t.rope_parameters.sliding_attention)?,
            rope_global: parse_rope(&t.rope_parameters.full_attention)?,
        })
    }
}
```

```rust
use std::collections::HashMap;
use serde_json::Value;

#[derive(Debug, Clone)]
pub struct QuantConfig {
    pub default_bits: i32,
    pub default_group_size: i32,
    /// 完整权重前缀 -> (bits, group_size)
    overrides: HashMap<String, (i32, i32)>,
}

impl QuantConfig {
    /// 兼容 `quantization` 与 `quantization_config` 两种键名;无量化段返回 Ok(None)。
    pub fn from_config_str(s: &str) -> Result<Option<Self>> {
        let v: Value = serde_json::from_str(s)?;
        let q = v.get("quantization").or_else(|| v.get("quantization_config"));
        let Some(q) = q else { return Ok(None) };
        let obj = q.as_object().ok_or_else(|| Error::Config("quantization not object".into()))?;

        let default_bits = obj.get("bits").and_then(|x| x.as_i64()).unwrap_or(4) as i32;
        let default_group_size = obj.get("group_size").and_then(|x| x.as_i64()).unwrap_or(64) as i32;

        let mut overrides = HashMap::new();
        for (k, val) in obj {
            // 仅形如 {bits, group_size} 的对象才是 per-module override
            if let Some(o) = val.as_object() {
                if let (Some(b), Some(g)) = (o.get("bits").and_then(|x| x.as_i64()),
                                             o.get("group_size").and_then(|x| x.as_i64())) {
                    if k != "mode" {
                        overrides.insert(k.clone(), (b as i32, g as i32));
                    }
                }
            }
        }
        Ok(Some(QuantConfig { default_bits, default_group_size, overrides }))
    }

    /// 返回 (bits, group_size)。prefix 为不含 `.weight/.scales/.biases` 的模块前缀。
    pub fn quant_for(&self, prefix: &str) -> (i32, i32) {
        self.overrides.get(prefix)
            .copied()
            .unwrap_or((self.default_bits, self.default_group_size))
    }
}
```

```rust
//! Gemma 4 RMSNorm: standard gamma (weight*norm) or no-scale (pure norm, for v_norm).
use mlx_rs::{Array, ops};
use mlx_rs::error::Exception;

pub struct GemmaRmsNorm {
    weight: Option<Array>, // None => with_scale=false
    eps: f32,
}

impl GemmaRmsNorm {
    /// with_scale=false:纯 RMSNorm,无权重(用于 v_norm)。
    pub fn new_no_scale(_dim: i32, eps: f32) -> Self {
        Self { weight: None, eps }
    }
    /// with_scale=true:标准 gamma,weight 来自 checkpoint(初始化为 ones)。
    pub fn from_weight(weight: Array, eps: f32) -> Self {
        Self { weight: Some(weight), eps }
    }

    pub fn forward(&self, x: &Array) -> Result<Array, Exception> {
        // normed = x * rsqrt(mean(x^2, last, keepdims) + eps)
        let var = ops::mean(&x.square()?, &[-1][..], true)?;
        let normed = x.multiply(&ops::rsqrt(&var.add(Array::from_f32(self.eps))?)?)?;
        match &self.weight {
            Some(w) => normed.multiply(w),
            None => Ok(normed),
        }
    }
}
```

> 注:具体 mlx-rs op 名（`ops::mean`/`ops::rsqrt`/`square`/`multiply`/`Array::from_f32`）以本仓库 mlx-rs 当前 API 为准,如签名不符,参照 `qwen3-mlx/src/model.rs` 中 RMSNorm/算子用法对齐。亦可直接复用 `mlx_rs::fast::rms_norm(x, weight, eps)`(with_scale 分支),no-scale 分支传入全 1 weight 或手算。

```rust
//! Gemma 4 dual RoPE.
//! - sliding layers: standard full-dim RoPE (use mlx_rs::nn::Rope, dims=head_dim).
//! - global layers: proportional RoPE — frequency denominator is FULL head_dim,
//!   only the first rotary_dim dims are rotated, the rest are identity (freq=0).

/// 返回长度 head_dim/2 的 inv_freq:前 rotary_dim/2 个按 theta^-(2i/head_dim),其余补 0。
pub fn proportional_inv_freq(head_dim: i32, rotary_dim: i32, theta: f32) -> Vec<f32> {
    let half = (head_dim / 2) as usize;
    let rot_half = (rotary_dim / 2) as usize;
    let mut out = vec![0.0f32; half];
    for i in 0..rot_half {
        let exponent = (2 * i) as f32 / head_dim as f32;
        out[i] = theta.powf(-exponent);
    }
    out
}
```

> M1 会基于此 `inv_freq` 构造 cos/sin 并对 q/k 前 `rotary_dim` 维施加旋转、余维透传。M0 只需此频率构造与下方 example 的逐元素对齐即可定论 R1。

```rust
//! Weight loading helpers (mirrors qwen3-mlx private helpers, not exported there).
use std::collections::{HashMap, HashSet};
use std::path::Path;
use mlx_rs::{Array, builder::Builder, module::Param, nn, module::Module};
use serde::Deserialize;
use crate::error::{Error, Result};

#[derive(Deserialize)]
struct WeightMap { weight_map: HashMap<String, String> }

pub fn load_all_weights(model_dir: &Path) -> Result<HashMap<String, Array>> {
    let idx = model_dir.join("model.safetensors.index.json");
    let json = std::fs::read_to_string(idx)?;
    let wm: WeightMap = serde_json::from_str(&json)?;
    let files: HashSet<&String> = wm.weight_map.values().collect();
    let mut all = HashMap::new();
    for f in files {
        let loaded = Array::load_safetensors(model_dir.join(f).to_str().unwrap())?;
        all.extend(loaded);
    }
    Ok(all)
}

pub fn weight_keys(model_dir: &Path) -> Result<Vec<String>> {
    let idx = model_dir.join("model.safetensors.index.json");
    let json = std::fs::read_to_string(idx)?;
    let wm: WeightMap = serde_json::from_str(&json)?;
    let mut keys: Vec<String> = wm.weight_map.into_keys().collect();
    keys.sort();
    Ok(keys)
}

pub fn get_weight(weights: &HashMap<String, Array>, key: &str) -> Result<Array> {
    weights.get(key).cloned().ok_or_else(|| Error::WeightNotFound(key.to_string()))
}

pub fn make_quantized_linear(weights: &HashMap<String, Array>, prefix: &str,
                             group_size: i32, bits: i32) -> Result<nn::QuantizedLinear> {
    let weight = get_weight(weights, &format!("{prefix}.weight"))?;
    let scales = get_weight(weights, &format!("{prefix}.scales"))?;
    let biases = get_weight(weights, &format!("{prefix}.biases"))?;
    let inner = nn::Linear { weight: Param::new(weight), bias: Param::new(None) };
    let mut ql = nn::QuantizedLinear { group_size, bits,
        scales: Param::new(scales), biases: Param::new(biases), inner };
    ql.freeze_parameters(true);
    Ok(ql)
}
```

```rust
//! Gemma 4 decoder layer: 4 layernorms, two residuals, trailing layer_scalar.
use mlx_rs::{Array, module::Module};
use mlx_rs::error::Exception;
use crate::attention::Attention;
use crate::mlp::Mlp;
use crate::norm::GemmaRmsNorm;

pub struct TransformerBlock {
    pub input_layernorm: GemmaRmsNorm,
    pub post_attention_layernorm: GemmaRmsNorm,
    pub pre_feedforward_layernorm: GemmaRmsNorm,
    pub post_feedforward_layernorm: GemmaRmsNorm,
    pub attn: Attention,
    pub mlp: Mlp,
    pub layer_scalar: Array, // 标量
}

impl TransformerBlock {
    pub fn forward(&mut self, x: &Array, mask: Option<&Array>) -> Result<Array, Exception> {
        let h = self.input_layernorm.forward(x)?;
        let h = self.attn.forward(&h, mask)?;
        let h = self.post_attention_layernorm.forward(&h)?;
        let x = x.add(&h)?; // residual 1
        let h2 = self.pre_feedforward_layernorm.forward(&x)?;
        let h2 = self.mlp.forward(&h2)?;
        let h2 = self.post_feedforward_layernorm.forward(&h2)?;
        let x = x.add(&h2)?; // residual 2
        x.multiply(&self.layer_scalar) // 层末 layer_scalar
    }
}
```
